cmake_minimum_required(VERSION 3.21)
project(URUIBridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(URUI_DEBUG "Compile native diagnostic probes, profiling and selftests" OFF)

# Build a self-contained plugin with static URUI and static FreeType.
# Packaged plugins expose the versioned production C ABI on every target.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static third-party dependencies" FORCE)
set(URUI_FONT_ENGINE freetype CACHE STRING "" FORCE)
set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)

# Versioned source ships with the package; no network or machine-local cache is needed.
add_subdirectory(ThirdParty/freetype freetype)
if(EMSCRIPTEN)
  # Visibility cannot isolate duplicate definitions in Wasm archives. Propagate the
  # same names to the font engine declarations and the vendored implementation.
  target_compile_options(freetype PUBLIC "-include" "${CMAKE_CURRENT_SOURCE_DIR}/freetype_symbols.h")
endif()
if(NOT TARGET Freetype::Freetype)
  add_library(Freetype::Freetype ALIAS freetype)
endif()
add_subdirectory(ThirdParty/gumbo gumbo)
add_subdirectory(Engine urui)
set(urui_engine_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Engine")

set(URUI_BRIDGE_SOURCES
  bridge_runtime.cpp
  bridge_documents.cpp
  bridge_elements.cpp
  bridge_events.cpp
  bridge_effects.cpp
  bridge_authoring.cpp
  bridge_render_plan.cpp
  document_registry.cpp
  render_plan_compiler.cpp
  style_properties.cpp
  render_program_parameters.cpp
)
if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_library(URUIBridge STATIC ${URUI_BRIDGE_SOURCES})
  set_target_properties(URUIBridge PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bridge")
else()
  add_library(URUIBridge SHARED ${URUI_BRIDGE_SOURCES})
endif()
# The upstream debugger is not part of a production Unity plugin.
target_link_libraries(URUIBridge PRIVATE URUI::Core)
target_compile_definitions(URUIBridge PRIVATE URUI_EXPORTS)
if(MSVC)
  target_compile_options(URUIBridge PRIVATE /W4 /utf-8)
else()
  target_compile_options(URUIBridge PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
set_target_properties(URUIBridge PROPERTIES OUTPUT_NAME URUIBridge)
if(WIN32)
  set_target_properties(URUIBridge PROPERTIES PREFIX "")
endif()
if(EMSCRIPTEN)
  # Preserve allocation-failure handling in the bridge and its dependencies.
  target_compile_options(URUIBridge PRIVATE -fexceptions)
  target_compile_options(urui_core PRIVATE -fexceptions)
endif()
if(URUI_DEBUG)
  target_sources(urui_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/Debug/diagnostics.cpp")
  target_sources(URUIBridge PRIVATE Debug/bridge_debug.cpp Debug/lookup_queue_tests.cpp)
  target_include_directories(URUIBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_include_directories(urui_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(URUIBridge PRIVATE URUI_DEBUG=1)
  target_compile_definitions(urui_core PRIVATE URUI_DEBUG=1)
endif()

if(EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(urui_plugin_archive "${CMAKE_CURRENT_BINARY_DIR}/libURUIBridge.a")
  add_custom_command(OUTPUT "${urui_plugin_archive}"
    COMMAND "${CMAKE_COMMAND}"
      "-DURUI_ARCHIVE=${urui_plugin_archive}"
      "-DURUI_AR=${CMAKE_AR}"
      "-DURUI_APPLE=${APPLE}"
      "-DURUI_BRIDGE=$<TARGET_FILE:URUIBridge>"
      "-DURUI_CORE=$<TARGET_FILE:urui_core>"
      "-DURUI_FREETYPE=$<TARGET_FILE:freetype>"
      "-DURUI_GUMBO=$<TARGET_FILE:gumbo>"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/bundle-static.cmake"
    DEPENDS URUIBridge urui_core freetype gumbo
      "${CMAKE_CURRENT_SOURCE_DIR}/bundle-static.cmake"
    VERBATIM)
  add_custom_target(URUIBridgePlugin ALL DEPENDS "${urui_plugin_archive}")
endif()
